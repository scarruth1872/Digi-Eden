class Player {
public:
    std::string name;
    int level;
    int experience;
    int health;
    int attack;
    int defense;

    Player(std::string name) : name(name), level(1), experience(0), health(100), attack(10), defense(5) {}

    void gainExperience(int exp) {
        experience += exp;
        while (experience >= level * 100) { // Example leveling system
            levelUp();
        }
    }

    void levelUp() {
        level++;
        health += 20;
        attack += 5;
        defense += 2;
        std::cout << name << " leveled up to " << level << "!\n";
    }

    void takeDamage(int damage) {
        int actualDamage = damage - defense;
        if (actualDamage < 0) actualDamage = 0;
        health -= actualDamage;
        if (health <= 0) {
            std::cout << name << " has been defeated!\n";
        }
    }
};

C++

// In include/core/quest.h
#ifndef QUEST_H
#define QUEST_H

#include <string>
#include <functional>
#include <fstream>
#include <sstream>

enum QuestType { SERVICE_TO_OTHERS, SERVICE_TO_SELF, BALANCED };

struct Quest {
    std::string name;
    std::string description;
    bool completed;
    int experienceReward;
    int serviceToOthersReward;
    int serviceToSelfReward;
    std::string completionMessage;
    QuestType type;
    std::string location;

    Quest() : completed(false), experienceReward(0), serviceToOthersReward(0), serviceToSelfReward(0), type(BALANCED) {}

    static Quest loadFromFile(const std::string& filename);
};

#endif

// In src/core/quest.cpp
#include "quest.h"
#include <iostream>

Quest Quest::loadFromFile(const std::string& filename) {
    Quest quest;
    std::ifstream file(filename);
    if (file.is_open()) {
        std::string line;
        while (std::getline(file, line)) {
            std::istringstream iss(line);
            std::string key;
            if (std::getline(iss, key, ':')) {
                std::string value;
                if (std::getline(iss, value)) {
                    // Trim whitespace from value
                    value.erase(0, value.find_first_not_of(" \t\n\r\f\v"));
                    value.erase(value.find_last_not_of(" \t\n\r\f\v") + 1);

                    if (key == "name") quest.name = value;
                    else if (key == "description") quest.description = value;
                    else if (key == "experienceReward") quest.experienceReward = std::stoi(value);
                    else if (key == "serviceToOthersReward") quest.serviceToOthersReward = std::stoi(value);
                    else if (key == "serviceToSelfReward") quest.serviceToSelfReward = std::stoi(value);
                    else if (key == "completionMessage") quest.completionMessage = value;
                    else if (key == "type") {
                        if (value == "serviceToOthers") quest.type = SERVICE_TO_OTHERS;
                        else if (value == "serviceToSelf") quest.type = SERVICE_TO_SELF;
                        else quest.type = BALANCED;
                    }
                    else if (key == "location") quest.location = value;
                }
            }
        }
        file.close();
    } else {
        std::cerr << "Unable to open quest file: " << filename << std::endl;
    }
    return quest;
}

                if (std::getline(iss, value)) {
                    // Trim whitespace from value
                    value.erase(0, value.find_first_not_of(" \t\n\r\f\v"));
                    value.erase(value.find_last_not_of(" \t\n\r\f\v") + 1);

                    if (key == "name") quest.name = value;
                    else if (key == "description") quest.description = value;
                    else if (key == "experienceReward") quest.experienceReward = std::stoi(value);
                    else if (key == "serviceToOthersReward") quest.serviceToOthersReward = std::stoi(value);
                    else if (key == "serviceToSelfReward") quest.serviceToSelfReward = std::stoi(value);
                    else if (key == "completionMessage") quest.completionMessage = value;
                    else if (key == "type") {
                        if (value == "serviceToOthers") quest.type = SERVICE_TO_OTHERS;
                        else if (value == "serviceToSelf") quest.type = SERVICE_TO_SELF;
                        else quest.type = BALANCED;
                    }
                    else if (key == "location") quest.location = value;
                    else if (key.find("objective") == 0){
                        std::string typeStr;
                        std::string target;
                        int quantity = 1;
                        std::istringstream objectiveIss(value);
                        std::getline(objectiveIss, typeStr, ',');
                        std::getline(objectiveIss, target, ',');
                        std::string quantityStr;
                        if(std::getline(objectiveIss, quantityStr)){
                            quantity = std::stoi(quantityStr);
                        }
                        QuestObjectiveType type;
                        if(typeStr == "LOCATION") type = LOCATION;
                        else if (typeStr == "COLLECT_ITEM") type = COLLECT_ITEM;
                        else if (typeStr == "DEFEAT_ENEMY") type = DEFEAT_ENEMY;
                        else {
                            std::cerr << "Invalid objective type: " << typeStr << std::endl;
                            continue;
                        }
                        quest.objectives.push_back(QuestObjective(type, target, quantity));
                    }
                }
            }
        }
        file.close();
    } else {
        std::cerr << "Unable to open quest file: " << filename << std::endl;
    }
    return quest;
}

    for(auto& enemy : enemies){
        std::cout << "A " << enemy.name << " engages " << player.name << "!\n";
        while (player.health > 0 && enemy.health > 0) {
            int playerDamage = player.attack;
            int enemyDamage = enemy.attack;

            // Add some randomness to damage
            playerDamage += getRandomNumber(-2, 2); // +/- 2 damage
            enemyDamage += getRandomNumber(-1, 1);   // +/- 1 damage

            if (playerDamage < 0) playerDamage = 0;
            if (enemyDamage < 0) enemyDamage = 0;

            enemy.takeDamage(playerDamage);
            if (enemy.health > 0) {

            std::cout << player.name << " Health: " << player.health << "\n";
            std::cout << enemy.name << " Health: " << enemy.health << "\n";
        }

        if (player.health > 0 && enemy.health <= 0) {
            player.gainExperience(50, false);
            defeatedEnemies.push_back(enemy.name);
        } else if (player.health <= 0){
            return defeatedEnemies;
        }
    }

            int playerDamage = player.attack;
            int enemyDamage = enemy.attack;

            // Add some randomness to damage
            playerDamage += getRandomNumber(-2, 2); // +/- 2 damage
            enemyDamage += getRandomNumber(-1, 1);   // +/- 1 damage

            if (playerDamage < 0) playerDamage = 0;
            if (enemyDamage < 0) enemyDamage = 0;

            enemy.takeDamage(playerDamage);
            if (enemy.health > 0) {
                if(enemy.type == EnemyType::RANGED){
                    if(abs(player.currentLocationIndex - enemy.currentLocationIndex) <= enemy.attackRange){
                        player.takeDamage(enemyDamage);
                    }
                    else{
                        std::cout << enemy.name << " is out of range.\n";
                    }
                }
                else{
                    player.takeDamage(enemyDamage);
                }
            }

            std::cout << player.name << " Health: " << player.health << "\n";
            std::cout << enemy.name << " Health: " << enemy.health << "\n";
        }

        if (player.health > 0 && enemy.health <= 0) {
            player.gainExperience(50, false);
            defeatedEnemies.push_back(enemy.name);
        } else if (player.health <= 0){
            return defeatedEnemies;
        }
    }
    return defeatedEnemies;

std::map<int, DialogueNode> loadDialogueFromFile(const std::string& filename) {
    std::map<int, DialogueNode> dialogue;
    std::ifstream file(filename);
    if (file.is_open()) {
        std::string line;
        int currentNode = -1;
        while (std::getline(file, line)) {
            if(line.find("options:") == 0){
                continue;
            }
            else if(line.find("    ") == 0){
                std::string optionText;
                int nextNode;
                std::istringstream iss(line);
                std::string dump;
                iss >> dump >> optionText;
                std::string delimiter = ":";
                size_t pos = 0;
                std::string token;
                while ((pos = optionText.find(delimiter)) != std::string::npos) {
                    token = optionText.substr(0, pos);
                    optionText.erase(0, pos + delimiter.length());
                }
                nextNode = std::stoi(optionText);
                dialogue[currentNode].nextNodes[token] = nextNode;
            }
            else{
                std::istringstream iss(line);
                std::string nodeStr;
                std::string text;
                std::getline(iss, nodeStr, ':');
                std::getline(iss, text);
                int node = std::stoi(nodeStr);
                dialogue[node].text = text;
                currentNode = node;
            }
        }
        file.close();
    } else {
        std::cerr << "Unable to open dialogue file: " << filename << std::endl;
    }
    return dialogue;
}

std::map<std::string, Location> loadLocations(const std::string& filename, std::vector<Item>& loadedItems) {
    std::map<std::string, Location> locations;
    std::ifstream file(filename);
    if (file.is_open()) {
        std::string line;
        int index = 0;
        while (std::getline(file, line)) {
            std::istringstream iss(line);
            std::string name;
            std::string description;
            std::string npcsString;
            std::string density4Description;
            std::string densityHint;
            std::string key;
            std::string value;
            std::string locationName;
            while(std::getline(iss, line)){
                std::istringstream lineStream(line);
                if(std::getline(lineStream, key, ':') && std::getline(lineStream, value)){
                    if(key == "name"){
                        locationName = value;
                        locations[locationName].locationIndex = index;
                        index++;
                    }
                    else if(key == "description"){
                        locations[locationName].description = value;
                    }
                    else if(key == "npcs"){
                        std::istringstream npcsIss(value);
                        std::string npc;
                        while(std::getline(npcsIss, npc, ',')){
                            locations[locationName].npcs.push_back(npc);
                        }
                    }
                    else if(key == "density4Description"){
                        locations[locationName].density4Description = value;
                    }
                    else if(key == "densityHint"){
                        locations[locationName].densityHint = value;
                    }
                    else if (key.find("object") == 0){
                        std::string name;
                        std::string description;
                        std::string itemName;
                        std::istringstream objectiveIss(value);
                        std::getline(objectiveIss, name, ',');
                        std::getline(objectiveIss, description, ',');
                        if(std::getline(objectiveIss, itemName)){
                            for(auto& loadedItem : loadedItems){
                                if(loadedItem.name == itemName){
                                    locations[locationName].objects.push_back({name, description, &loadedItem, nullptr});
                                    break;
                                }
                            }
                        }
                        else{
                            locations[locationName].objects.push_back({name, description, nullptr, nullptr});
                        }
                    }
                    else if (key.find("environment_effect") == 0) {
                        int density;
                        std::string effectDescription;
                        std::istringstream objectiveIss(value);
                        objectiveIss >> density;
                        std::getline(objectiveIss >> std::ws, effectDescription);
                        locations[locationName].environmentEffects.push_back({density, effectDescription, [](Player& player){}});
                    }
                }
            }
        }
        file.close();
    } else {
        std::cerr << "Unable to open location file: " << filename << std::endl;
    }
    return locations;

void loadGame(Player& player, std::vector<Quest>& quests, std::vector<Item>& loadedItems, const std::string& filename) {
    std::ifstream file(filename);
    if (file.is_open()) {
        std::string line;

        // Load Player
        std::getline(file, line); // "Player" block start
        if (line == "Player") {
            player.load(file, loadedItems);
        }

        // Load Quests
        std::getline(file, line); // "Quests" block start
        if (line == "Quests") {
            std::getline(file, line);
            int numQuests = std::stoi(line);
            quests.clear(); // Clear existing quests before loading
            for (int i = 0; i < numQuests; ++i) {
                std::getline(file, line); // "Quest" block start
                if (line == "Quest") {
                    Quest quest;
                    quest.load(file);
                    quests.push_back(quest);
                }
            }
            std::getline(file, line); // Consume "EndQuests"
        }
        file.close();
        std::cout << "Game loaded successfully.\n";
    } else {
        std::cerr << "Error loading game. File not found or corrupted.\n";
    }
}
